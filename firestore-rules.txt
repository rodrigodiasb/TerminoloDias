rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isValidString(field, min, max) {
      return field is string && field.size() >= min && field.size() <= max;
    }

    function isValidResult() {
      return isValidString(request.resource.data.playerName, 1, 40)
        && isValidString(request.resource.data.playerGroup, 1, 40)
        && request.resource.data.score is number
        && request.resource.data.score >= 0
        && request.resource.data.correct is number
        && request.resource.data.correct >= 0
        && request.resource.data.total is number
        && request.resource.data.total >= 0
        && request.resource.data.percent is number
        && request.resource.data.percent >= 0
        && request.resource.data.percent <= 100;
    }

    function isValidReport() {
      return request.resource.data.questionId is number
        && isValidString(request.resource.data.playerName, 1, 40)
        && isValidString(request.resource.data.playerGroup, 1, 40)
        && isValidString(request.resource.data.reportText, 5, 700);
    }

    match /resultados/{docId} {
      allow read: if true;
      allow create: if isValidResult();
      allow update, delete: if false;
    }

    match /recursos_questoes/{docId} {
      allow read: if false;
      allow create: if isValidReport();
      allow update, delete: if false;
    }
  }
}
